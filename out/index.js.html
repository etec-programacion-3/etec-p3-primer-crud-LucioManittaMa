import express from 'express';
import bodyParser from 'body-parser';
import { Sequelize, Model, DataTypes } from 'sequelize';
import { config } from 'dotenv';

const app = express();
const port = 3000;

config();
const filename = "database.db";
console.log(filename);
const sequelize = new Sequelize({
    dialect: 'sqlite',
    storage: filename
});

/**
 * Book model definition
 * @class Book
 * @extends {Model}
 */
class Book extends Model { }
Book.init({
    /**
     * @type {string}
     * @memberof Book
     */
    title: DataTypes.STRING,
    /**
     * @type {string}
     * @memberof Book
     */
    author: DataTypes.STRING,
    /**
     * @type {string}
     * @memberof Book
     */
    isbn: DataTypes.STRING,
    /**
     * @type {string}
     * @memberof Book
     */
    editorial: DataTypes.STRING,
    /**
     * @type {number}
     * @memberof Book
     */
    pages: DataTypes.INTEGER
}, { sequelize, modelName: 'book' });

sequelize.sync();

app.use(bodyParser.urlencoded({ extended: false }));
app.use(bodyParser.json());

/**
 * GET /books
 * @summary Retrieve all books
 * @returns {Array<Book>} 200 - An array of books
 */
app.get('/books', async (req, res) => {
    const books = await Book.findAll();
    res.json(books);
});

/**
 * GET /books/{id}
 * @summary Retrieve a book by ID
 * @param {string} id.path.required - Book ID
 * @returns {Book} 200 - A book object
 * @returns {object} 404 - Book not found
 */
app.get('/books/:id', async (req, res) => {
    const book = await Book.findByPk(req.params.id);
    if (book) {
        res.json(book);
    } else {
        res.status(404).json({ message: 'Book not found' });
    }
});

/**
 * POST /books
 * @summary Create a new book
 * @param {Book} request.body.required - Book object
 * @returns {Book} 200 - The created book object
 */
app.post('/books', async (req, res) => {
    const book = await Book.create(req.body);
    res.json(book);
});

/**
 * PUT /books/{id}
 * @summary Update a book by ID
 * @param {string} id.path.required - Book ID
 * @param {Book} request.body.required - Book object
 * @returns {Book} 200 - The updated book object
 * @returns {object} 404 - Book not found
 */
app.put('/books/:id', async (req, res) => {
    const book = await Book.findByPk(req.params.id);
    if (book) {
        await book.update(req.body);
        res.json(book);
    } else {
        res.status(404).json({ message: 'Book not found' });
    }
});

/**
 * DELETE /books/{id}
 * @summary Delete a book by ID
 * @param {string} id.path.required - Book ID
 * @returns {object} 200 - Deletion message
 * @returns {object} 404 - Book not found
 */
app.delete('/books/:id', async (req, res) => {
    const book = await Book.findByPk(req.params.id);
    if (book) {
        await book.destroy();
        res.json({ message: 'Book deleted' });
    } else {
        res.status(404).json({ message: 'Book not found' });
    }
});

app.listen(port, () => {
    console.log(`Server listening on port ${port}`)
});
